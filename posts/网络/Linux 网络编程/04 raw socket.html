04 raw socket

在Linux操作系统中，原始套接字（Raw Sockets）允许程序直接发送和接收网络层的原始数据包,甚至是连接层的帧

原始套接字可以用于多种目的，包括：

  • 网络诊断：例如，使用原始套接字可以编写类似ping的工具，直接发送和接收ICMP（Internet Control Message Protocol）数据包。
  • 实现自定义协议：开发者可以在原始套接字的基础上实现自己的网络协议。
  • 网络安全：原始套接字可以用于创建网络监控工具，例如嗅探器（sniffer），来捕获和分析网络上的数据包。
  • 路由和转发：程序可以使用原始套接字来处理和转发数据包，实现自定义的网络路由功能。

由于原始套接字能够访问和操作网络层/连接层的原始数据，因此它们的使用通常需要超级用户权限，并且可能存在安全风险。不正确的使用原始套接字可能会影响网络的安全性和稳定性。

在Linux中，要创建一个原始套接字，可以使用socket系统调用，并将套接字类型指定为SOCK_RAW。此外，还需要指定一个协议，如IPPROTO_TCP、IPPROTO_UDP或IPPROTO_ICMP等，以指示原始套接字应该处理哪种类型的网络数据包。

下面是一个使用原始套接字发送ICMP ECHO REQUEST的例子

#include <stdio.h> 
#include <stdlib.h> 
#include <string.h> 
#include <unistd.h> 
#include <sys/types.h> 
#include <sys/socket.h> 
#include <netinet/in.h> 
#include <netinet/ip.h> 
#include <netinet/ip_icmp.h> 
#include <netdb.h> 
#include <arpa/inet.h>

#define ICMP_ECHO 8
#define ICMP_ECHOREPLY 0
#define ICMP_HEADER_LEN 8


unsigned short checksum(void *buffer, int size) {
    unsigned short *buf = (unsigned short *) buffer;
    unsigned long cksum=0;
    while(size > 1) {
        cksum += *buf++;
        size -= sizeof(unsigned short);
    }
    if(size)
        cksum += *(unsigned char*)buf;
    cksum = (cksum >> 16) + (cksum & 0xffff);
    cksum += (cksum >> 16);
    return (unsigned short)(~cksum);
}

int main(int argc, char *argv[]) {
    struct icmp icmp_hdr;
    char packet[4096];
    struct protoent *proto;
    struct sockaddr_in dest;
    int sockfd;

    proto = getprotobyname("ICMP");
    sockfd = socket(AF_INET, SOCK_RAW, proto->p_proto);
    if (sockfd < 0) {
        perror("Socket error");
        return 1;
    }

    memset(&dest, 0, sizeof(dest));
    dest.sin_family = AF_INET;
    dest.sin_addr.s_addr = inet_addr("180.101.50.188");

    memset(&icmp_hdr, 0, sizeof(icmp_hdr));
    icmp_hdr.icmp_type = ICMP_ECHO;
    icmp_hdr.icmp_cksum = 0;
    icmp_hdr.icmp_seq = 0;
    icmp_hdr.icmp_id = getpid() & 0xffff;
    icmp_hdr.icmp_cksum = checksum(&icmp_hdr, ICMP_HEADER_LEN);

    memcpy(packet, &icmp_hdr, ICMP_HEADER_LEN);

    if (sendto(sockfd, packet, ICMP_HEADER_LEN, 0, (struct sockaddr*)&dest, sizeof(dest)) < 0) {
        perror("Sendto error");
        return 1;
    }

    struct sockaddr_in from;
    socklen_t fromlen = sizeof(from);
    if (recvfrom(sockfd, packet, sizeof(packet), 0, (struct sockaddr*)&from, &fromlen) < 0) {
        perror("Recvfrom error");
        return 1;
    }

    memcpy(&icmp_hdr, packet, ICMP_HEADER_LEN);
    printf("type %d\n",icmp_hdr.icmp_type);

    if (icmp_hdr.icmp_type == ICMP_ECHOREPLY) {
        printf("PING reply received!\n");
    }

    return 0;
}

下面是一个使用原始套接字发送数据帧的例子

#include <sys/socket.h>
#include <linux/if_packet.h>
#include <net/ethernet.h>
#include <arpa/inet.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>

#define BUF_SIZE 1500

int main() {
    int raw_sock;
    char buffer[BUF_SIZE];
    struct sockaddr_ll sa;

    /* Create the raw socket */
    raw_sock = socket(AF_PACKET, SOCK_RAW, htons(ETH_P_ALL));

    if (raw_sock == -1) {
        /* Error occurred */
        perror("Error creating socket");
        return -1;
    }

    /* Set up sockaddr_ll structure */
    memset(&sa, 0, sizeof(struct sockaddr_ll));
    sa.sll_family = AF_PACKET;
    sa.sll_protocol=htons(ETH_P_ALL);
    sa.sll_ifindex = if_nametoindex("eth0");  // Sending through eth0 interface. Change it as per your system.
    sa.sll_halen = ETH_ALEN;
    sa.sll_addr[0] = 0x00;  // Destination MAC. Change it to the destination MAC address.
    sa.sll_addr[1] = 0x11;
    sa.sll_addr[2] = 0x22;
    sa.sll_addr[3] = 0x33;
    sa.sll_addr[4] = 0x44;
    sa.sll_addr[5] = 0x55;

    /* Create a dummy Ethernet frame */
    memset(buffer, 'A', BUF_SIZE);

    /* Send the Ethernet frame */
    if(sendto(raw_sock, buffer, BUF_SIZE, 0, (struct sockaddr*)&sa, sizeof(struct sockaddr_ll)) == -1) {
        perror("Error sending frame");
        return -1;
    }

    /* Close the raw socket */
    close(raw_sock);

    return 0;
}

创建原始socket的接口 socket(AF_INET, SOCK_RAW, PROTOCOL), 其第三个参数指定了套接字使用的协议。对于Internet地址族的原始套接字，这个参数可以是以下之一：

  • socket.IPPROTO_IP (0): 用于接收所有IP层的数据包, 如ICMP。
  • socket.IPPROTO_ICMP (1): 用于专门处理ICMP协议的数据包，例如ping请求和回复。
  • socket.IPPROTO_TCP (6): 用于处理TCP协议的数据包。
  • socket.IPPROTO_UDP (17): 用于处理UDP协议的数据包。
  • socket.IPPROTO_RAW (255): 用于发送裸的IP数据包，允许用户完全控制IP头部，包括设置TTL、协议版本等。
  • etc

在使用原始套接字时，你需要选择合适的协议值，以便能够接收或发送特定类型的数据包。如果你想要处理所有IP层的数据包，可以使用socket.IPPROTO_IP。如果你只对ICMP数据包感兴趣，那么应该使用socket.IPPROTO_ICMP。

如果是发送连接层的数据帧，则需要将地址族设置为AF_PACKET，其第三个协议部分可以是

  • ETH_P_IP: Used to capture IP packets.
  • ETH_P_IPV6: Used to capture IPv6 packets.
  • ETH_P_ARP: Used to capture ARP packets.
  • ETH_P_RARP: Used to capture RARP packets.
  • ETH_P_ALL: Used to capture all types of packets.
