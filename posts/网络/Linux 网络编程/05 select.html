05 select

select, epoll, poll 是I/O多路复用的三种常见机制。它们允许单个进程/线程监视多个文件描述符（通常是网络套接字）的可读、可写和异常事件，而无需为每个文件描述符单独创建一个线程或进程处理。I/O多路复用与非阻塞套接字使用，提高单个进程/线程的并发处理能力。

select: select是最早出现的I/O多路复用机制。 它通过一个select函数来监视文件描述符集合，等待其中一个或多个文件描述符变为“就绪”状态。

#include <sys/select.h>

int select(int nfds, fd_set *readfds, fd_set *writefds, fd_set *exceptfds, struct timeval *timeout);

//nfds: This is the highest-numbered file descriptor in any of the three sets, plus 1.
//readfds: An optional pointer to a set of file descriptors to be checked for being ready to read.
//writefds: An optional pointer to a set of file descriptors to be checked for being ready to write.
//exceptfds: An optional pointer to a set of file descriptors to be checked for error conditions.
//timeout: An optional timeout value. select returns if this amount of time passes with no file descriptors becoming ready. 
//The function returns the total number of bits set in the bit masks. In case of timeout, it will return 0 and it will return -1 for errors.

timeout: The timeout parameter in the select function determines how long select should wait for an event to occur on one of the file descriptors it's monitoring.

If timeout is set to 0 (both tv_sec and tv_usec in the timeval structure are 0), select will not block. It will check all the file descriptors and return immediately. This is useful for polling.

If timeout is set to NULL, select will block indefinitely until an event occurs on one of the file descriptors it's monitoring.

In both cases, select returns the number of file descriptors contained in the three returned descriptor sets (those that are ready for some kind of I/O) or -1 if an error occurred.

fd_set:文件描述符数组，每个bit可表示为一个文件描述符,一个数组最多可设置1024个文件描述符

#define __FD_SETSIZE        1024

fd_set的使用

/* Access macros for `fd_set'.  */
#define    FD_SET(fd, fdsetp)    __FD_SET (fd, fdsetp)
#define    FD_CLR(fd, fdsetp)    __FD_CLR (fd, fdsetp)
#define    FD_ISSET(fd, fdsetp)    __FD_ISSET (fd, fdsetp)
#define    FD_ZERO(fdsetp)        __FD_ZERO (fdsetp)

下面是一个简单的使用select的tcp服务器，当客户端连接上时，会给其发送个welcome message. 后续echo客户端消息。该服务器是个单线程，我们最多支持1000个并发的客户端。(due to limitation of 1024 on fd_set)。如果并发的客户端超过限制，我们就会发送一个消息后断开连接，prevent unnecessary resource usage and potential issues with unmanaged connections.

#include <arpa/inet.h>
#include <errno.h>
#include <fcntl.h>
#include <netinet/in.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/select.h>
#include <sys/socket.h>
#include <sys/types.h>
#include <unistd.h>

#define MAX_CLIENTS 10
#define BUFFER_SIZE 1024

void set_socket_non_blocking(int socketfd) {
    int flags = fcntl(socketfd, F_GETFL, 0);
    if (flags == -1) {
        perror("fcntl");
        exit(EXIT_FAILURE);
    }

    if (fcntl(socketfd, F_SETFL, flags | O_NONBLOCK) == -1) {
        perror("fcntl");
        exit(EXIT_FAILURE);
    }
}

int main(int argc, char *argv[]) {
    fd_set master_fd_set, copy_fd_set;
    int socketfd, addrlen, new_socket, client_sockets[MAX_CLIENTS], activity, i, valread, sd;
    int max_sd;
    struct sockaddr_in address;
    char buffer[BUFFER_SIZE];
    char hellomessage[] = "Hello from server";
    char fullmessage[] = "Server is full. Try again later";
    bool full;

    for (i = 0; i < MAX_CLIENTS; i++) {
        client_sockets[i] = -1;
    }

    if ((socketfd = socket(AF_INET, SOCK_STREAM, 0)) <= 0) {
        perror("socket failed");
        exit(EXIT_FAILURE);
    }

    //INADDR_ANY is a special IP address in network programming, particularly in the context of setting up server sockets. It's a symbolic constant used in socket programming to bind the socket to all available interfaces. It translates to 0.0.0.0 in IPv4, allowing the socket to listen for client requests coming from any interface on the machine.
    //if we want to change to 127.0.0.1 on lo interface, we should to this
    /*
    struct sockaddr_in server_address;
    server_address.sin_family = AF_INET;
    server_address.sin_port = htons(8080);
    server_address.sin_addr.s_addr = inet_addr("127.0.0.1");
    */
    //This makes the server listen only on the localhost interface
    address.sin_family = AF_INET;
    address.sin_addr.s_addr = INADDR_ANY;
    address.sin_port = htons(8888);

    //htons: Stands for "Host TO Network Short". It converts a short integer from host byte order to network byte order (which is Big Endian).

    //ntohs: Stands for "Network TO Host Short". It converts a short integer from network byte order (Big Endian) to host byte order.

    //inet_ntoa: Stands for "Internet Network TO ASCII". It converts an Internet network address into an ASCII string in Internet standard dotted format.

    //inet_addr: Converts an IPv4 Internet network address in the ASCII string format into binary (network byte order) format.

    if (bind(socketfd, (struct sockaddr *)&address, sizeof(address)) < 0) {
        perror("bind failed");
        exit(EXIT_FAILURE);
    }

    if (listen(socketfd, 3) < 0) {
        perror("listen");
        exit(EXIT_FAILURE);
    }

    addrlen = sizeof(address);
    puts("Waiting for connections ...");

    set_socket_non_blocking(socketfd);
    FD_ZERO(&master_fd_set);
    FD_SET(socketfd, &master_fd_set);
    max_sd = socketfd;

    while (1) {
        //每次的select循环之前，copy_fd_set需要被设置好。在select返回后，copy_fd_set只设置有事件发生的fd的bit,其他的bit被清空
        copy_fd_set = master_fd_set;

        activity = select(max_sd + 1, &copy_fd_set, NULL, NULL, NULL);
        if ((activity < 0) && (errno != EINTR)) {
            printf("select error");
        }

        if (FD_ISSET(socketfd, &copy_fd_set)) {
            if ((new_socket = accept(socketfd, (struct sockaddr *)&address, (socklen_t *)&addrlen)) < 0) {
                perror("accept");
                exit(EXIT_FAILURE);
            }

            printf("New connection , socket fd is %d , ip is : %s , port : %d\n", new_socket, inet_ntoa(address.sin_addr), ntohs(address.sin_port));
            if (send(new_socket, hellomessage, strlen(hellomessage), 0) != strlen(hellomessage)) {
                perror("send");
            }

            puts("Welcome message sent successfully");

            for (i = 0; i < MAX_CLIENTS; i++) {
                full=true;
                if (client_sockets[i] == -1) {
                    client_sockets[i] = new_socket;
                    printf("Adding to list of sockets as %d\n", i);

                    set_socket_non_blocking(new_socket);
                    FD_SET(new_socket, &master_fd_set);
                    if (new_socket > max_sd)
                        max_sd = new_socket;
                    full=false;
                    break;
                }
            }
            if(full=true){
                    send(new_socket, fullmessage, strlen(fullmessage), 0);
                    close(new_socket);
            }

        }

        for (i = 0; i < MAX_CLIENTS; i++) {
            sd = client_sockets[i];

            if (FD_ISSET(sd, &copy_fd_set)) {
                if ((valread = read(sd, buffer, BUFFER_SIZE)) == 0) {
                    getpeername(sd, (struct sockaddr *)&address, (socklen_t *)&addrlen);
                    printf("Host disconnected , ip %s , port %d \n", inet_ntoa(address.sin_addr), ntohs(address.sin_port));
                    close(sd);
                    FD_CLR(sd, &master_fd_set);
                    client_sockets[i] = -1;
                }else if(valread <0){
                    perror("read error");
                    close(sd);
                    FD_CLR(sd, &master_fd_set);
                    client_sockets[i] = -1;
                }
                else {
                    buffer[valread] = '\0';
                    send(sd, buffer, strlen(buffer), 0);
                }
            }
        }
    }

    return 0;
}

select的优点就是其是最早的I/O复用机制，大多数 Unix-like operating systems都支持。 In Windows, the equivalent functionality is provided by the select function in the Winsock API. 其缺点就是它支持的文件描述符数量有限（通常受限于FD_SETSIZE），且每次调用都需要重新设置数组并传入。
