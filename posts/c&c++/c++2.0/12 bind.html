其中_Placeholder是个空模板类

而模板参数_Num的作用，在于我们可以通过如下方法获取其是第几个参数

  template<typename _Tp>

    : public integral_constant<int, 0>
    { };

    struct is_placeholder<_Placeholder<_Num> >
    : public integral_constant<int, _Num>
    { };

  template<int _Num>

比如

integral_constant，可在编译期创建任何整数类型的常量

  template<typename _Tp, _Tp __v>
    struct integral_constant
    {
      static constexpr _Tp                  value = __v;
      typedef _Tp                           value_type;
      typedef integral_constant<_Tp, __v>   type;
      constexpr operator value_type() const noexcept { return value; }
      constexpr value_type operator()() const noexcept { return value; }

bind的简单实现如下

#include <iostream>
#include <functional> //std::function, std::forward, std::move  std::make_index_sequence std::index_sequence
#include <tuple> //std::tuple
#include <type_traits> //std::integral_constant enable_if_t is_placeholder remove_reference_t remove_pointer_t



template<size_t n>
using index_constant = std::integral_constant<size_t, n>;

template<typename ... Args>
class binder_list
{
public:
    //虽然这里使用了万能引用，但并不影响最终bind()函数中的参数是被拷贝到tuple中的，因为tuple的模板参数并不是引用类型。
    template<typename ... TArgs>
    constexpr binder_list(TArgs&&... args) noexcept
    : boundedArgs_{std::forward<TArgs>(args)...}
    {}

    //此接口从tuple中获取对应的参数，比如被如下` operator[](index_constant<2>{})`调用，则实例化为 decltype(auto) operator[](index_constant<2>){ return std::get<2>(boundedArgs_);},返回tuple中的第三个参数，也就是 placeholder::_1. 注意std::get<>的起始索引是0
    template<size_t n>
    constexpr decltype(auto) operator[](index_constant<n>)  noexcept
    {
        return std::get<n>(boundedArgs_);
    }

private:
    std::tuple<Args...> boundedArgs_;    //这里的tuple就是我们在调用bind时，传递的完整参数列表。 比如 auto bf = bind(f, 1, 2, placeholder::_1), 此时这里的tuple 就是<1,2, placeholder::_1>.
    //可见bind中传递的参数是拷贝的
};

template<typename ... Args>
class callee_list
{
public:
    template<typename ... TArgs>
    constexpr callee_list(TArgs&&... args)  noexcept 
    : boundedArgs_{std::forward<TArgs>(args)...}
    {}

    //如果我们从binder_list中获得的参数是被绑定的参数，比如从<1,2, placeholder::_1> 获取第一个参数或者第二个参数，假设为t,  通过callee_list()[t]调用时，将实例化这个接口，该接口直接返回这个参数的引用
    template<typename T, std::enable_if_t<(std::is_placeholder<std::remove_reference_t<T>>::value == 0)>* = nullptr>
    constexpr decltype(auto) operator[](T&& t) noexcept
    {
        return std::forward<T>(t);
    }
    //如果我们从binder_list中获得的参数是placeholder，比如从<1,2, placeholder::_1> 获取第三个参数，假设为t,  通过callee_list()[t]调用时，将实例化这个接口，该接口返回binder调用时对应的参数，也就是从boundedArgs_中获取第一个参数 std::get<0>(boundedArgs_)
    template<typename T, std::enable_if_t<(std::is_placeholder<T>::value != 0)>* = nullptr>
    constexpr decltype(auto) operator[](T) noexcept
    {
        return std::get<std::is_placeholder<T>::value - 1>(boundedArgs_);
    }

private:
    std::tuple<Args&&...> boundedArgs_;     //这里的tuple存储的是我们在调用 binder时，传递的参数。 比如 `auto bf = bind(f, 1, 2, placeholder::_1); bf(10);` 则tuple中存储为 <10>. 注意binder在被调用时，参数是引用传递的，并不是拷贝。因为这里使用的是万能引用，最终tuple中存储的都是引用类型
};

template<typename Fn, typename ... Args>
class binder
{
public:
    template<typename TFn, typename ... TArgs>
    constexpr binder(TFn&& f, TArgs&&... args) noexcept
    : f_{std::forward<TFn>(f)},
      argumentList_{std::forward<TArgs>(args)...}
    {}


    template<typename ... CallArgs>
    constexpr decltype(auto) operator()(CallArgs&&... args)  noexcept
    {
        //std::make_index_sequence<sizeof...(Args)>{} 返回  std::index_sequence,具体见后文
        return call(std::make_index_sequence<sizeof...(Args)>{}, std::forward<CallArgs>(args)...);
    }

private:
    template<typename ... CallArgs, size_t ... Seq>
    constexpr decltype(auto) call(std::index_sequence<Seq...>, CallArgs&&... args)  noexcept
    {   //callee_list<CallArgs...>{std::forward<CallArgs>(args)...} 实例化 callee_list 对象 ，然后对该对象调用operator[], 其参数是argumentList_[index_constant<Seq>{}]
        //也就是 argumentList_[index_constant<0>{}],argumentList_[index_constant<1>{}],argumentList_[index_constant<2>{}]
        //也就是依次获取出 binder_list 中的数据，<1,2,placeholder::_1>, 当获取到placeholder::_1时，callee_list 的operator[] 返回的时callee_list中存储的第一个参数。
        //这样 f_的调用就变成了 f_(1,2,10);
        return f_((callee_list<CallArgs...>{std::forward<CallArgs>(args)...}[argumentList_[index_constant<Seq>{}]])...);
    }
private:
    std::function<std::remove_reference_t<std::remove_pointer_t<Fn>>> f_;
    binder_list<Args...> argumentList_;
};

namespace me
{
template<typename Fn, typename ... Args>
binder<Fn, Args...> bind(Fn&& f, Args&&... args)
{
    return binder<Fn, Args...>{std::forward<Fn>(f), std::forward<Args>(args)...};
}
}

int func(int a, int b,int c){
    return a+b + c ;
}

int main(){
    auto bf = me::bind(func,1,2, std::placeholders::_1);
    std::cout<<bf(10)<<std::endl;
}

bind函数返回的binder对象，里面存储一个std::function对象，一个bind调用时的参数列表(binder_list)，参数是拷贝的。binder在调用时，会创建临时的callee_list对象，存储调用binder时传递的参数列表，只不过这个参数是引用传递的。

在callee_list中的两个operator重载

    template<typename T, std::enable_if_t<(std::is_placeholder<std::remove_reference_t<T>>::value == 0)>* = nullptr>
    decltype(auto) operator[](T&& t)
    {
        return std::forward<T>(t);
    }

    template<typename T, std::enable_if_t<(std::is_placeholder<T>::value != 0)>* = nullptr>
    decltype(auto) operator[](T)
    {
        return std::get<std::is_placeholder<T>::value - 1>(boundedArgs_);
    }

如果我们调用operator[]时传递的是实际参数，则 std::is_placeholder<std::remove_reference_t<T>>::value == 0为true, std::is_placeholder<T>::value != 0为false. 而std::enable_if_t，只在模板参数为true时，才是有效的，其实现如下

  template<bool _Cond, typename _Tp = void>
    using enable_if_t = typename enable_if<_Cond, _Tp>::type;

  // Primary template.
  /// Define a member typedef `type` only if a boolean constant is true.
  template<bool, typename _Tp = void>
    struct enable_if
    { };

  // Partial specialization for true.
  template<typename _Tp>
    struct enable_if<true, _Tp>
    { typedef _Tp type; };

关于 index_sequence

   /// Alias template index_sequence
  template<size_t... _Idx>

    struct integer_sequence
    {
      typedef _Tp value_type;
      static constexpr size_t size() noexcept { return sizeof...(_Idx); }
    };

关于make_index_sequence

  /// Alias template make_index_sequence
  template<size_t _Num>
    using make_index_sequence = make_integer_sequence<size_t, _Num>;


  /// Alias template make_integer_sequence
  template<typename _Tp, _Tp _Num>
    using make_integer_sequence
    //Clang defines __make_integer_seq for this purpose.
#if __has_builtin(__make_integer_seq)
      = __make_integer_seq<integer_sequence, _Tp, _Num>;
#else
    //For GCC and other compilers, use __integer_pack instead.
      = integer_sequence<_Tp, __integer_pack(_Num)...>;

这个__integer_pack应该是gcc自己实现的helper function, 返回{0，1，2..._Num-1}的序列包

一个简单的lambda,这个简单的实现不支持placeholder, 只能完全的参数绑定

template <typename Func, typename... Args>
auto bind(Func&& f, Args&&... args)
{
    return [&f, &args...]() -> decltype(auto)
    {
        return f(args...);
    };
}
