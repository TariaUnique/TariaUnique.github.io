12 观察者模式

Observer Design Pattern: Define a one-to-many dependency between objects so that when one object changes state, all its dependents are notified and updated automatically.

最经典的实现方式

#include <memory>
#include <list>
#include <algorithm>
#include <iostream>

using std::shared_ptr;
using std::list;

class Message{};

class Observer;

class  Subject {
public:
  virtual ~Subject(){}
  virtual void registerObserver(shared_ptr<Observer> observer)=0;
  virtual void removeObserver(shared_ptr<Observer> observer)=0;
  virtual void notifyObservers(const Message& message)=0;
};

class Observer {
public:
  virtual ~Observer(){}
  virtual void update(const Message&  message)=0;
};

class ConcreteSubject : public  Subject {
public:
  void registerObserver(shared_ptr<Observer> observer){
    observers.push_back(observer);
  }
  void removeObserver(shared_ptr<Observer> observer){
    auto it = std::find_if(observers.begin(),observers.end(),[&observer](const shared_ptr<Observer>& regOb){return regOb.get() == observer.get();});
    if(it!=observers.end()){
        observers.erase(it);
    }
  }
  void notifyObservers(const Message& message){
    for(auto& observer: observers){
        observer->update(message);
    }
  }
private:
  list<shared_ptr<Observer>> observers;
};

class ConcreteObserverOne : public  Observer {
public:
  void update(const Message&  message) {
    //TODO: 获取消息通知，执行自己的逻辑...
    std::cout<<"ConcreteObserverOne is notified."<<std::endl;
  }
};

class ConcreteObserverTwo : public  Observer {
public:
  void update(const Message&  message) {
    //TODO: 获取消息通知，执行自己的逻辑...
    std::cout<<"ConcreteObserverTwo is notified."<<std::endl;
  }
};

int main(){
    shared_ptr<Subject> subject = std::make_shared<ConcreteSubject>();
    shared_ptr<Observer> obOne = std::make_shared<ConcreteObserverOne>();
    shared_ptr<Observer> obTwo = std::make_shared<ConcreteObserverTwo>();

    subject->registerObserver(obOne);
    subject->registerObserver(obTwo);
    subject->notifyObservers(Message());
    subject->removeObserver(obTwo);
    subject->notifyObservers(Message());
    return 0;
}

一个实例： 假设我们在开发一个 P2P 投资理财系统，用户注册成功之后，我们会给用户发放投资体验金。代码实现大致是下面这个样子的：

#include <memory>
#include <string>

using std::shared_ptr;
using std::string;

class UserService{
public:
  long reg(string telephone, string password){
    //...
    return 0x1000;
  }
};

class PromotionService{
public:
  void issueNewUserExperienceCash(long userId){
    //...
  }
};

class UserController {
public:
  UserController(shared_ptr<UserService> _userService, shared_ptr<PromotionService> _promotionService):
    userService(_userService),
    promotionService(_promotionService){}

  long reg(string telephone, string password) {
    //省略输入参数的校验代码
    //省略userService.register()异常的try-catch代码
    long userId = userService->reg(telephone, password);
    promotionService->issueNewUserExperienceCash(userId);
    return userId;
  }
private:
  shared_ptr<UserService> userService; // 依赖注入
  shared_ptr<PromotionService> promotionService;// 依赖注入
};

虽然注册接口做了两件事情，注册和发放体验金，违反单一职责原则，但是，如果没有扩展和修改的需求，现在的代码实现是可以接受的。如果非得用观察者模式，就需要引入更多的类和更加复杂的代码结构，反倒是一种过度设计。相反，如果需求频繁变动，比如，用户注册成功之后，不再发放体验金，而是改为发放优惠券，并且还要给用户发送一封“欢迎注册成功”的站内信。这种情况下，我们就需要频繁地修改 register() 函数中的代码，违反开闭原则。而且，如果注册成功之后需要执行的后续操作越来越多，那 register() 函数的逻辑会变得越来越复杂，也就影响到代码的可读性和可维护性。这个时候，观察者模式就能派上用场了。利用观察者模式，我对上面的代码进行了重构。重构之后的代码如下所示：

#include <memory>
#include <string>
#include <list>

using std::shared_ptr;
using std::string;
using std::list;


class  RegObserver {
public:
  virtual ~RegObserver(){}
  virtual void handleRegSuccess(long userId)=0;
};

class PromotionService{
public:
  void issueNewUserExperienceCash(long userId){
    //...
  }
};

class RegPromotionObserver : public  RegObserver {
public:
  RegPromotionObserver(shared_ptr<PromotionService> _promotionService):
    promotionService(_promotionService){}
  void handleRegSuccess(long userId){
    promotionService->issueNewUserExperienceCash(userId);
  }
private:
  shared_ptr<PromotionService> promotionService; // 依赖注入
};

class NotificationService{
public:
  void sendInboxMessage(long userId,string message){
    //...
  }
};


class RegNotificationObserver : public  RegObserver {
public:
  RegNotificationObserver(shared_ptr<NotificationService> _notificationService):
    notificationService(_notificationService){}
  void handleRegSuccess(long userId){
    notificationService->sendInboxMessage(userId, "Welcome...");
  }
private:
  shared_ptr<NotificationService> notificationService; // 依赖注入
};




class UserService{
public:
  long reg(string telephone, string password){
    //...
    return 0x1000;
  }
};

class UserController {
public:
  UserController(shared_ptr<UserService> _userService):
    userService(_userService){}

  void setRegObservers(const list<shared_ptr<RegObserver>>& observers) {
    for( auto & ob: observers){
        regObservers.push_back(ob);
    }
  }
  long reg(string telephone, string password) {
    //省略输入参数的校验代码
    //省略userService.register()异常的try-catch代码
    long userId = userService->reg(telephone, password);

    for (auto&  observer : regObservers) {
      observer->handleRegSuccess(userId);
    }

    return userId;
  }
private:
  shared_ptr<UserService> userService; // 依赖注入
  list<shared_ptr<RegObserver>> regObservers; 

};

观察者模式的应用场景非常广泛，小到代码层面的解耦，大到架构层面的系统解耦，再或者一些产品的设计思路，都有这种模式的影子，比如，邮件订阅、RSS Feeds，本质上都是观察者模式。不同的应用场景和需求下，这个模式也有截然不同的实现方式，有同步阻塞的实现方式，也有异步非阻塞的实现方式；有进程内的实现方式，也有跨进程的实现方式。 上述的实现是一种同步阻塞的实现方式。观察者和被观察者代码在同一个线程内执行，被观察者一直阻塞，直到所有的观察者代码都执行完成之后，才执行后续的代码。对照上面讲到的用户注册的例子，register() 函数依次调用执行每个观察者的 handleRegSuccess() 函数，等到都执行完成之后，才会返回结果给客户端。

如果注册接口是一个调用比较频繁的接口，对性能非常敏感，希望接口的响应时间尽可能短，那我们可以将同步阻塞的实现方式改为异步非阻塞的实现方式，以此来减少响应时间。如下,我们直接使用boost asio库的threadpool 和io_context来异步执行观察者的代码。

#include <memory>
#include <string>
#include <list>
#include <thread>
#include <boost/asio.hpp>

using std::shared_ptr;
using std::string;
using std::list;


class  RegObserver {
public:
  virtual ~RegObserver(){}
  virtual void handleRegSuccess(long userId)=0;
};

class PromotionService{
public:
  void issueNewUserExperienceCash(long userId){
    //...
  }
};

class RegPromotionObserver : public  RegObserver {
public:
  RegPromotionObserver(shared_ptr<PromotionService> _promotionService):
    promotionService(_promotionService){}
  void handleRegSuccess(long userId){
    promotionService->issueNewUserExperienceCash(userId);

  }
private:
  shared_ptr<PromotionService> promotionService; // 依赖注入
};

class NotificationService{
public:
  void sendInboxMessage(long userId,string message){
    //...
  }
};


class RegNotificationObserver : public  RegObserver {
public:
  RegNotificationObserver(shared_ptr<NotificationService> _notificationService):
    notificationService(_notificationService){}
  void handleRegSuccess(long userId){
    notificationService->sendInboxMessage(userId, "Welcome...");

  }
private:
  shared_ptr<NotificationService> notificationService; // 依赖注入
};




class UserService{
public:
  long reg(string telephone, string password){
    //...
    return 0x1000;
  }
};

class UserController {
using io_context = boost::asio::io_context;
public:
  UserController(shared_ptr<UserService> _userService, io_context & _io):
    userService(_userService),io(_io){}

  void setRegObservers(const list<shared_ptr<RegObserver>>& observers) {
    for( auto & ob: observers){
        regObservers.push_back(ob);
    }
  }
  long reg(string telephone, string password) {
    //省略输入参数的校验代码
    //省略userService.register()异常的try-catch代码
    long userId = userService->reg(telephone, password);

    for (auto&  observer : regObservers) {
      io.post([&](){
        observer->handleRegSuccess(userId);
      });
    }
    return userId;
  }
private:
  shared_ptr<UserService> userService; // 依赖注入
  list<shared_ptr<RegObserver>> regObservers; 
  io_context& io;
};

int main(){
    //...
    boost::asio::io_context io;
    shared_ptr<UserService> userService = std::make_shared<UserService>();
    shared_ptr<UserController> controller = std::make_shared<UserController>(userService,io);
    //....
    //4个observer线程执行observer的代码
    boost::asio::thread_pool pool(4);
    boost::asio::post(pool, [&]() { io.run(); });
    pool.join();
    //...
}

不管是同步阻塞实现方式还是异步非阻塞实现方式，都是进程内的实现方式。如果用户注册成功之后，我们需要发送用户信息给大数据征信系统，而大数据征信系统是一个独立的系统，跟它之间的交互是跨不同进程的，那如何实现一个跨进程的观察者模式呢？如果大数据征信系统提供了发送用户注册信息的 RPC 接口，我们仍然可以沿用之前的实现思路，在 handleRegSuccess() 函数中调用 RPC 接口来发送数据。但是，我们还有更加优雅、更加常用的一种实现方式，那就是基于消息队列（Message Queue，比如 ActiveMQ）来实现。当然，这种实现方式也有弊端，那就是需要引入一个新的系统（消息队列），增加了维护成本。不过，它的好处也非常明显。在原来的实现方式中，观察者需要注册到被观察者中，被观察者需要依次遍历观察者来发送消息。而基于消息队列的实现方式，被观察者和观察者解耦更加彻底，两部分的耦合更小。被观察者完全不感知观察者，同理，观察者也完全不感知被观察者。被观察者只管发送消息到消息队列，观察者只管从消息队列中读取消息来执行相应的逻辑。

观察者模式的应用场景非常广泛，小到代码层面的解耦，大到架构层面的系统解耦，再或者一些产品的设计思路，都有这种模式的影子，比如，邮件订阅、RSS Feeds，本质上都是观察者模式。不同的应用场景和需求下，这个模式也有截然不同的实现方式，有同步阻塞的实现方式，也有异步非阻塞的实现方式；有进程内的实现方式，也有跨进程的实现方式。
